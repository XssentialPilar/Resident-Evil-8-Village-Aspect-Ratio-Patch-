cmake_minimum_required(VERSION 3.20)
project(RE8DirectStorageCompatibility LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This compatibility probe requires Windows x64")
endif()
if(NOT EXISTS "${DIRECTSTORAGE_INCLUDE_DIR}/dstorage.h")
    message(FATAL_ERROR "Set DIRECTSTORAGE_INCLUDE_DIR to the locally extracted preview headers")
endif()
add_executable(RE8DirectStorageProbe main.cpp)
target_compile_features(RE8DirectStorageProbe PRIVATE cxx_std_20)
target_include_directories(RE8DirectStorageProbe PRIVATE "${DIRECTSTORAGE_INCLUDE_DIR}")
target_compile_definitions(RE8DirectStorageProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    set_property(TARGET RE8DirectStorageProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(RE8DirectStorageProbe PRIVATE /W4 /EHsc)
endif()

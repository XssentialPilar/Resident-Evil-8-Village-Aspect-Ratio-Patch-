cmake_minimum_required(VERSION 3.20)
project(RE8DirectStorageBridge LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 is required")
endif()
if(NOT EXISTS "${DIRECTSTORAGE_INCLUDE_DIR}/dstorage.h")
    message(FATAL_ERROR "DIRECTSTORAGE_INCLUDE_DIR must identify the locally reviewed SDK headers")
endif()
add_library(RE8DirectStorage SHARED RE8DirectStorageBridge.cpp)
add_library(RE8DirectStorageFaultTest SHARED RE8DirectStorageBridge.cpp)
target_compile_definitions(RE8DirectStorageFaultTest PRIVATE RE8_DS_TESTING=1)
add_executable(RE8DirectStorageHarness RE8DirectStorageBridge.cpp)
set_target_properties(RE8DirectStorageHarness PROPERTIES OUTPUT_NAME re8)
target_compile_definitions(RE8DirectStorageHarness PRIVATE RE8_DS_HARNESS=1)
foreach(target IN ITEMS RE8DirectStorage RE8DirectStorageFaultTest RE8DirectStorageHarness)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE ../../external/reframework-sdk "${DIRECTSTORAGE_INCLUDE_DIR}")
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(${target} PRIVATE /W4 /EHsc)
endforeach()

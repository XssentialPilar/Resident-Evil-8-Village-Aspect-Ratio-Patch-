cmake_minimum_required(VERSION 3.20)
project(RE8AgilityBridge VERSION 1.0.0 LANGUAGES CXX ASM_MASM)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "RE8AgilityBridge requires Windows x64/MSVC.")
endif()
if(NOT BRIDGE_GENERATED_DIR)
    message(FATAL_ERROR "Use tools/RE8AgilityBridge.Build.ps1 to generate the exact system export table.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(RE8AgilityBridge SHARED RE8AgilityBridge.cpp
    "${BRIDGE_GENERATED_DIR}/forwarders.asm" "${BRIDGE_GENERATED_DIR}/d3d12.def")
set_target_properties(RE8AgilityBridge PROPERTIES OUTPUT_NAME d3d12 PREFIX "")
target_compile_features(RE8AgilityBridge PRIVATE cxx_std_20)
target_compile_definitions(RE8AgilityBridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_include_directories(RE8AgilityBridge PRIVATE "${BRIDGE_GENERATED_DIR}")
target_link_libraries(RE8AgilityBridge PRIVATE dxguid)
target_compile_options(RE8AgilityBridge PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /permissive- /EHsc>)
add_executable(RE8AgilityBridgeTestHost RE8AgilityBridgeTestHost.cpp)
target_compile_features(RE8AgilityBridgeTestHost PRIVATE cxx_std_20)
target_include_directories(RE8AgilityBridgeTestHost PRIVATE "${BRIDGE_GENERATED_DIR}")
target_compile_definitions(RE8AgilityBridgeTestHost PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(RE8AgilityBridgeTestHost PRIVATE dxgi dxguid)
target_compile_options(RE8AgilityBridgeTestHost PRIVATE /W4 /permissive- /EHsc)

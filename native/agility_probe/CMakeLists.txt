cmake_minimum_required(VERSION 3.20)
project(RE8AgilityProbe VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "RE8AgilityProbe requires Windows x64.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(RE8AgilityProbe RE8AgilityProbe.cpp)
target_compile_features(RE8AgilityProbe PRIVATE cxx_std_20)
target_compile_definitions(RE8AgilityProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(RE8AgilityProbe PRIVATE dxgi)
if(MSVC)
    target_compile_options(RE8AgilityProbe PRIVATE /W4 /permissive- /EHsc)
endif()

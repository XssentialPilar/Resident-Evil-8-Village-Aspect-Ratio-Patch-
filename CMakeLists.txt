cmake_minimum_required(VERSION 3.20)
project(RE8VillagePatchTests LANGUAGES CXX)
enable_testing()

foreach(test_name IN ITEMS re8_display_geometry_tests re8_cursor_policy_tests re8_camera_projection_tests re8_map_drag_tests)
    add_executable(${test_name} "tests/${test_name}.cpp")
    target_include_directories(${test_name} PRIVATE reframework_src/src/mods)
    target_compile_features(${test_name} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(${test_name} PRIVATE /EHsc /UNDEBUG)
    else()
        target_compile_options(${test_name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()

# Reuse the exact Lua runtime built with the dependency; never fetch another one.
set(RE8_LUA_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/reframework_src/build/Release/lua.lib" CACHE FILEPATH "Existing REFramework Lua library")
if(EXISTS "${RE8_LUA_LIBRARY}")
    add_executable(re8_lua_test_runner tests/re8_lua_test_runner.cpp)
    target_include_directories(re8_lua_test_runner PRIVATE reframework_src/dependencies/lua/src)
    target_link_libraries(re8_lua_test_runner PRIVATE "${RE8_LUA_LIBRARY}")
    if(MSVC)
        set_property(TARGET re8_lua_test_runner PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
        target_compile_options(re8_lua_test_runner PRIVATE /EHsc)
    endif()
    add_test(NAME re8_display_diagnostics_tests COMMAND re8_lua_test_runner
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/re8_display_diagnostics_tests.lua"
        "${CMAKE_CURRENT_SOURCE_DIR}/runtime/reframework/autorun/re8_display.lua")
else()
    message(FATAL_ERROR "Build the REFramework dependency first with tools/RE8Patch.Build.ps1; its Lua runtime is required for the complete test suite.")
endif()
